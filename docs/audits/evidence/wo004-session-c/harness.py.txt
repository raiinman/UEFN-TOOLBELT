Imports the reviewed client.py and mcp_server.py from the checkout, verifies
their identities against the mandate, and wraps each module's transport seam
(_direct_opener) in this process to record every connection attempt: client,
command, start/end times, and the HTTP status or exception type. It never
records request headers, the handoff contents, or any token.

Usage: py -3.13 harness.py <subcommand> [args]   (see SUBCOMMANDS at bottom)

def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def append(path: Path, record: dict) -> None:
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(record, sort_keys=True) + "\n")


def load_state() -> dict:
    return json.loads(STATE.read_text(encoding="utf-8")) if STATE.exists() else {}


def save_state(state: dict) -> None:
    STATE.write_text(json.dumps(state, indent=2, sort_keys=True), encoding="utf-8")


# ── environment guards ────────────────────────────────────────────────────────

def guard_environment() -> None:
    present = [k for k in FORBIDDEN_ENV if os.environ.get(k) or os.environ.get(k.lower())]
    if present:
        sys.exit("STOP: environment variables set: " + ", ".join(present))
    for name, want in IDENTITIES.items():
        got = sha256(REPO / name)
        if got != want:
            sys.exit(f"STOP: {name} sha256 {got} != mandate identity {want}")


def listening_ports() -> list[int]:
    out = subprocess.run(["netstat", "-ano", "-p", "TCP"], capture_output=True,
                         text=True, check=True).stdout
    ports = set()
    for line in out.splitlines():
        parts = line.split()
        if len(parts) >= 4 and parts[3] == "LISTENING":
            port = int(parts[1].rsplit(":", 1)[1])
            if 8765 <= port <= 8770:
                ports.add(port)
    return sorted(ports)


# ── loading the reviewed clients with the attempt recorder ────────────────────

class _FakeFastMCP:
    """Stand-in for mcp.server.fastmcp.FastMCP, as tests/test_mcp_security.py
    uses: tool() registers and returns the function unchanged, run() refuses.
    The `mcp` package is not installed; nothing under test depends on it."""

    def __init__(self, *args, **kwargs):
        self.tools = {}

    def tool(self):
        def register(function):
            self.tools[function.__name__] = function
            return function
        return register

    def run(self):
        raise AssertionError("FastMCP.run() must not run in the harness")


class _RecordingOpener:
    def __init__(self, real, label):
        self._real, self._label = real, label

    def open(self, req, timeout=None):
        try:
            command = json.loads(req.data.decode("utf-8")).get("command")
        except Exception:
            command = "<unparsed>"
        record = {"run": RUN, "client": self._label, "command": command,
                  "url_port": req.full_url.rsplit(":", 1)[1].split("/")[0],
                  "timeout": timeout, "start": time.time()}
        try:
            response = self._real.open(req, timeout=timeout)
        except urllib.error.HTTPError as exc:
            record.update(end=time.time(), http_status=exc.code, raised="HTTPError")
            append(ATTEMPTS, record)
            raise
        except BaseException as exc:
            reason = getattr(exc, "reason", None)
            record.update(end=time.time(), raised=type(exc).__name__,
                          reason=type(reason).__name__ if reason is not None else None)
            append(ATTEMPTS, record)
            raise
        record.update(end=time.time(), http_status=response.status, raised=None)
        append(ATTEMPTS, record)
        return response


def _wrap(module, label):
    real_factory = module._direct_opener

    def recording_factory():
        return _RecordingOpener(real_factory(), label)

    module._direct_opener = recording_factory


def load_clients():
    guard_environment()
    fastmcp = types.ModuleType("mcp.server.fastmcp")
    fastmcp.FastMCP = _FakeFastMCP
    sys.modules.setdefault("mcp", types.ModuleType("mcp"))
    sys.modules.setdefault("mcp.server", types.ModuleType("mcp.server"))
    sys.modules["mcp.server.fastmcp"] = fastmcp
    modules = {}
    for name in ("client", "mcp_server"):
        spec = importlib.util.spec_from_file_location("wo004_" + name, REPO / (name + ".py"))
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        _wrap(module, "client.py" if name == "client" else "mcp_server.py")
        modules[name] = module
    return modules["client"], modules["mcp_server"]


def attempts_for(run: str, since: float) -> list[dict]:
    if not ATTEMPTS.exists():
        return []
    rows = [json.loads(line) for line in ATTEMPTS.read_text(encoding="utf-8").splitlines() if line]
    return [r for r in rows if r["run"] == run and r["start"] >= since]


def call(label: str, fn) -> dict:
    """Run one client call; record result or exception with its timing."""
    t0 = time.time()
    try:
        result = fn()
        return {"call": label, "start": t0, "end": time.time(), "ok": True,
                "result": result, "attempts": attempts_for(RUN, t0)}
    except BaseException as exc:
        return {"call": label, "start": t0, "raised_at": time.time(), "ok": False,
                "exception": type(exc).__name__, "mro": [c.__name__ for c in type(exc).__mro__],
                "message": str(exc), "attempts": attempts_for(RUN, t0)}


def show(outcome: dict) -> None:
    n = len(outcome["attempts"])
    if outcome["ok"]:
        r = outcome["result"]
        if isinstance(r, str):
            try:
                r = json.loads(r)
            except ValueError:
                pass
        brief = {k: r[k] for k in ("status", "port", "transport", "uptime_s") if isinstance(r, dict) and k in r} or r
        print(f"  {outcome['call']}: OK in {outcome['end'] - outcome['start']:.3f}s, attempts={n}, result={brief}")
    else:
        first = outcome["message"].splitlines()[0] if outcome["message"] else ""
        print(f"  {outcome['call']}: {outcome['exception']} after {outcome['raised_at'] - outcome['start']:.3f}s,"
              f" attempts={n} | {first}")
    for a in outcome["attempts"]:
        print(f"    attempt {a['client']} {a['command']} port={a['url_port']} "
              f"{a['start']:.3f}->{a['end']:.3f} status={a.get('http_status')} raised={a.get('raised')} reason={a.get('reason')}")


def record_case(case: str, verdict: str, detail: dict) -> None:
    append(CASES, {"run": RUN, "case": case, "verdict": verdict, "recorded_at": time.time(), **detail})
    print(f"==> {case}: {verdict}")


def history(client_mod) -> dict:
    return client_mod.ToolbeltClient()._send("history", {"tail": 200})

def cmd_preflight(args):
    guard_environment()
    deployed = TOOL_TEST / BRIDGE_REL
    stamp = TOOL_TEST / "Content/Python/UEFN_Toolbelt/_build_stamp.json"
    repo_tree = REPO / "Content/Python/UEFN_Toolbelt"
    dep_tree = TOOL_TEST / "Content/Python/UEFN_Toolbelt"
    mismatched, missing = [], []
    for path in repo_tree.rglob("*"):
        if path.is_file() and "__pycache__" not in path.parts:
            rel = path.relative_to(repo_tree)
            other = dep_tree / rel
            if not other.exists():
                missing.append(str(rel))
            elif sha256(other) != sha256(path):
                mismatched.append(str(rel))
    info = {
        "client_py_sha256": sha256(REPO / "client.py"),
        "mcp_server_py_sha256": sha256(REPO / "mcp_server.py"),
        "repo_bridge_sha256": sha256(REPO / BRIDGE_REL),
        "deployed_bridge_sha256": sha256(deployed) if deployed.exists() else None,
        "build_stamp": json.loads(stamp.read_text(encoding="utf-8")) if stamp.exists() else None,
        "deployed_tree_mismatched": mismatched, "deployed_tree_missing": missing,
        "handoff_exists": HANDOFF.exists(), "ports_8765_8770_listening": listening_ports(),
        "python": sys.version.split()[0],
    }
    info["bridge_match"] = info["repo_bridge_sha256"] == info["deployed_bridge_sha256"]
    append(CASES, {"run": RUN, "case": "preflight", "recorded_at": time.time(), **info})
    for k, v in info.items():
        print(f"  {k}: {v}")


def cmd_ping(args):
    """Positive control / restart confirmation: one ping through the named client(s)."""
    c, s = load_clients()
    out = []
    if args[0] in ("client", "both"):
        out.append(call("client.py ping", lambda: c.ToolbeltClient().ping()))
    if args[0] in ("server", "both"):
        out.append(call("mcp_server.py ping", lambda: json.loads(s.ping())))
    for o in out:
        show(o)
    case = args[1] if len(args) > 1 else "ping"
    ok = all(o["ok"] and len(o["attempts"]) == 1 for o in out)
    ports = sorted({o["result"].get("port") for o in out if o["ok"]})
    if ok:
        state = load_state()
        state["port"] = ports[0] if len(ports) == 1 else ports
        save_state(state)
    record_case(case, "PASS" if ok and len(ports) == 1 else "FAIL", {"outcomes": out, "ports": ports})

def _window(before: dict, after: dict) -> list[dict]:
    delta = after["total"] - before["total"]
    return after["entries"][-delta:] if delta > 0 else []

def _watch_log(offset: int, pattern: re.Pattern, deadline: float):
    with EDITOR_LOG.open("rb") as f:
        f.seek(offset)
        buf = b""
        while time.time() < deadline:
            chunk = f.read()
            if chunk:
                buf += chunk
                m = pattern.search(buf.decode("utf-8", "replace"))
                if m:
                    return time.time(), float(m.group(1)), f.tell() - len(buf) + m.end()
            else:
                time.sleep(0.02)
    return None


START = re.compile(r"WO-004 stall start ([0-9]+\.[0-9]+)")
END = re.compile(r"WO-004 stall end ([0-9]+\.[0-9]+)")


def cmd_stall(args):
    """Arm a stall case; fires its request as soon as the stall-start print appears."""
    case = args[0]   # case4-client | case5-client | case7-server
    c, s = load_clients()
    state = load_state()
    if case == "case5-client" and "case5" not in state:
        sys.exit("STOP: run case5-baseline first")
    offset = EDITOR_LOG.stat().st_size
    armed = time.time()
    print(f"ARMED {case} at {armed:.3f}; waiting up to 600 s for the stall-start print", flush=True)
    seen = _watch_log(offset, START, armed + 600)
    if seen is None:
        record_case(case, "NOT RUN", {"reason": "no stall-start print within 600 s", "armed": armed})
        return
    seen_at, stall_start, pos = seen
    if case == "case4-client":
        fire = lambda: c.ToolbeltClient().ping()
    elif case == "case7-server":
        fire = lambda: s.ping()
    else:
        target = state["case5"]["target"]
        fire = lambda: c.ToolbeltClient()._send("set_viewport_camera", {"location": target})
    outcome = call(case + " request", fire)
    show(outcome)
    print("  waiting for the stall-end print ...", flush=True)
    ended = _watch_log(pos, END, time.time() + 180)
    stall_end = ended[1] if ended else None
    attempt = outcome["attempts"][0] if len(outcome["attempts"]) == 1 else None
    raised = outcome.get("raised_at")
    timing = {
        "stall_start_print": stall_start, "stall_end_print": stall_end,
        "log_seen_at": seen_at, "attempt_start": attempt["start"] if attempt else None,
        "client_raised_at": raised,
        "stall_began_before_attempt": bool(attempt) and stall_start < attempt["start"],
        "raised_before_stall_end": bool(raised) and stall_end is not None and raised < stall_end,
        "no_200_for_attempt": all(a.get("http_status") != 200 for a in outcome["attempts"]),
    }
    valid = all(timing[k] for k in ("stall_began_before_attempt", "raised_before_stall_end", "no_200_for_attempt"))
    expected = "TimeoutError" if case == "case7-server" else "CommandTimeout"
    checks = {
        "one_attempt": len(outcome["attempts"]) == 1,
        "unknown_outcome_class": not outcome["ok"] and outcome["exception"] == expected,
        "unknown_outcome_wording": not outcome["ok"] and UNKNOWN in outcome["message"],
        "arrival": ("bridge 504" if attempt and attempt.get("http_status") == 504
                    else "socket timeout" if attempt and attempt.get("raised") in ("TimeoutError", "timeout", "URLError")
                    else "other"),
    }
    print("  timing:", {k: v for k, v in timing.items()})
    print("  checks:", checks)
    detail = {"outcome": outcome, "timing": timing, "checks": checks}
    if case == "case5-client":
        if stall_end is None:
            record_case(case, "PENDING", {**detail, "note": "stall end not seen; run case5-verify"})
            return
        time.sleep(max(0.0, stall_end + 5 - time.time()))
        state = load_state()
        state["case5"]["fired"] = detail
        save_state(state)
        return _case5_verify(c, state, detail, valid)
    verdict = ("VOID" if not valid else
               "PASS" if checks["one_attempt"] and checks["unknown_outcome_class"] and checks["unknown_outcome_wording"]
               else "FAIL")
    record_case(case, verdict, detail)

def cmd_case5_baseline(args):
    c, s = load_clients()
    cam = call("get_viewport_camera (baseline)", lambda: c.ToolbeltClient()._send("get_viewport_camera"))
    show(cam)
    hist = call("history (baseline)", lambda: history(c))
    show(hist)
    if not (cam["ok"] and hist["ok"]):
        sys.exit("STOP: baseline reads failed")
    state = load_state()
    loc = cam["result"]["location"]
    target = [round(loc["x"] + 400.0, 1), round(loc["y"] + 400.0, 1), round(loc["z"] + 200.0, 1)]
    initial = state.get("camera_initial") or cam["result"]

    save_state(state)
    print(f"  initial camera kept for cleanup: {initial}")
    print(f"  target location: {target}; baseline history total={hist['result']['total']}")
    append(CASES, {"run": RUN, "case": "case5-baseline", "recorded_at": time.time(),
                   "camera": cam, "history": hist, "target": target})


def _case5_verify(c, state, fired, valid):
    base = state["case5"]
    h = call("history (verify)", lambda: history(c))
    show(h)
    cam = call("get_viewport_camera (verify)", lambda: c.ToolbeltClient()._send("get_viewport_camera"))
    show(cam)
    window = _window(base["baseline_history"], h["result"]) if h["ok"] else None
    sets = [e for e in (window or []) if e.get("command") == "set_viewport_camera"]
    loc = cam["result"]["location"] if cam["ok"] else None
    target = base["target"]
    checks = dict(fired["checks"])
    checks.update({
        "exactly_one_new_set_viewport_camera": len(sets) == 1,
        "total_below_500_both_reads": base["baseline_history"]["total"] < 500 and h["ok"] and h["result"]["total"] < 500,
        "camera_at_target": bool(loc) and all(abs(loc[k] - t) < 1.0 for k, t in zip("xyz", target, strict=True)),
    })
    print("  history window since baseline:", window)
    print("  checks:", checks)
    ok = (checks["one_attempt"] and checks["unknown_outcome_class"] and checks["unknown_outcome_wording"]
          and checks["exactly_one_new_set_viewport_camera"] and checks["total_below_500_both_reads"]
          and checks["camera_at_target"])
    verdict = "VOID" if not valid else ("PASS" if ok else "FAIL")
    record_case("case5-client", verdict, {**fired, "verify_history": h, "verify_camera": cam,
                                          "window": window, "checks": checks})

def cmd_unknown_command(args):
    which = args[0]   # client (case6) | server (case7)
    c, s = load_clients()
    name = "wo004_session_c_no_such_command"
    if which == "client":
        o = call("client.py _send unknown", lambda: c.ToolbeltClient()._send(name))
        ok = (not o["ok"] and o["exception"] == "ToolbeltError"
              and o["message"].startswith(f"UEFN reported a failure for command '{name}'")
              and "Unknown command" in o["message"])
        case = "case6-reported-failure"
    else:
        o = call("mcp_server.py _send unknown", lambda: s._send(name))
        ok = (not o["ok"] and o["exception"] == "RuntimeError"
              and o["message"].startswith(f"UEFN error for '{name}'")
              and "may have failed before running or partway through" in o["message"])
        case = "case7-server-reported-failure"
    show(o)
    one = len(o["attempts"]) == 1 and o["attempts"][0].get("http_status") == 200
    record_case(case, "PASS" if ok and one else "FAIL", {"outcome": o, "one_attempt_200": one})


def cmd_camera_restore(args):
    c, s = load_clients()
    initial = load_state().get("camera_initial")
    if not initial:
        sys.exit("nothing to restore: no initial camera recorded")
    loc = [initial["location"][k] for k in "xyz"]
    rot = [initial["rotation"][k] for k in ("pitch", "yaw", "roll")]
    o = call("set_viewport_camera (restore)", lambda: c.ToolbeltClient()._send(
        "set_viewport_camera", {"location": loc, "rotation": rot}))
    show(o)
    cam = call("get_viewport_camera (after restore)", lambda: c.ToolbeltClient()._send("get_viewport_camera"))
    show(cam)
    ok = cam["ok"] and all(abs(cam["result"]["location"][k] - initial["location"][k]) < 1.0 for k in "xyz") \
        and all(abs(cam["result"]["rotation"][k] - initial["rotation"][k]) < 0.5 for k in ("pitch", "yaw", "roll"))
    print(f"  initial: {initial}\n  now:     {cam.get('result')}")
    record_case("cleanup-camera-restore", "PASS" if ok else "FAIL", {"set": o, "get": cam, "initial": initial})
